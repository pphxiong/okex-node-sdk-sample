### **一、核心止盈止损策略**

#### **1. 动态止盈止损算法**
```javascript
class ProfitManager {
  constructor(config) {
    this.config = config;
  }

  // 计算止损价格
  calculateStopLoss(entryPrice, side) {
    const slPercent = this.getDynamicSL();
    return side === 'long' 
      ? entryPrice * (1 - slPercent)
      : entryPrice * (1 + slPercent);
  }

  // 计算止盈价格
  calculateTakeProfit(entryPrice, side) {
    const tpPercent = this.getDynamicTP();
    return side === 'long'
      ? entryPrice * (1 + tpPercent)
      : entryPrice * (1 - tpPercent);
  }

  // 动态止损比例
  getDynamicSL() {
    const baseSL = this.config.riskControl.stopLoss;
    const volatility = this.calculateVolatility();
    return Math.min(0.05, baseSL * (1 + volatility * 2)); // 最大止损5%
  }

  // 动态止盈比例
  getDynamicTP() {
    const baseTP = this.config.riskControl.takeProfit;
    const volatility = this.calculateVolatility();
    return Math.min(0.1, baseTP * (1 + volatility * 1.5)); // 最大止盈10%
  }

  // 计算市场波动率
  calculateVolatility() {
    const closes = this.marketData['1m'].slice(-14).map(c => c.close);
    return (Math.max(...closes) - Math.min(...closes)) / Math.min(...closes);
  }
}
```

### **三、策略实现代码**

#### **1. 订单执行模块**
```javascript
class OrderExecutor {
  constructor(exchange, config) {
    this.exchange = exchange;
    this.config = config;
    this.profitManager = new ProfitManager(config);
  }

  async executeOrder(side, entryPrice) {
    try {
      const size = await this.calculatePositionSize(entryPrice);
      const sl = this.profitManager.calculateStopLoss(entryPrice, side);
      const tp = this.profitManager.calculateTakeProfit(entryPrice, side);

      const order = await this.exchange.createOrder(
        this.config.symbol,
        'market',
        side,
        size,
        null,
        {
          stopLoss: { 
            type: 'stop_market',
            stopPrice: sl,
            reduceOnly: true
          },
          takeProfit: {
            type: 'take_profit_market',
            stopPrice: tp,
            reduceOnly: true
          }
        }
      );

      console.log(`订单执行成功: ${side} ${size} @ ${entryPrice}`);
      return order;

    } catch (error) {
      console.error('订单执行失败:', error);
      throw error;
    }
  }

  async calculatePositionSize(entryPrice) {
    const balance = await this.exchange.fetchBalance();
    const equity = balance.USDT.free;
    const riskAmount = equity * this.config.riskControl.maxLoss;
    return riskAmount / (entryPrice * this.config.riskControl.stopLoss);
  }
}
```

#### **2. 风险监控模块**
```javascript
class RiskMonitor {
  constructor(config) {
    this.config = config;
    this.dailyLoss = 0;
  }

  checkDailyLoss(profit) {
    this.dailyLoss += profit;
    if (this.dailyLoss < -this.config.riskControl.maxLoss) {
      console.error('触发最大日亏损限制，停止交易');
      process.exit(1);
    }
  }

  checkPositionRisk(position, currentPrice) {
    const unrealizedPnl = this.calculateUnrealizedPnl(position, currentPrice);
    if (unrealizedPnl < -position.entryPrice * this.config.riskControl.stopLoss * 1.5) {
      console.warn('触发额外风控，强制平仓');
      return true;
    }
    return false;
  }

  calculateUnrealizedPnl(position, currentPrice) {
    return position.side === 'long'
      ? (currentPrice - position.entryPrice) * position.size
      : (position.entryPrice - currentPrice) * position.size;
  }
}
```
