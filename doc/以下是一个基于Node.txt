const ccxt = require('ccxt')
const talib = require('talib')
const axios = require('axios')

// 配置模块
const config = {
  symbol: 'DOGE/USDT',
  timeframe: '15m',
  riskPerTrade: 0.02, // 单笔风险2%
  maxDailyLoss: 0.05, // 最大日亏损5%
  emaPeriods: [9, 21, 50],
  rsiPeriod: 14,
  atrPeriod: 14,
  exchange: new ccxt.binance({
    apiKey: process.env.API_KEY,
    secret: process.env.API_SECRET,
    enableRateLimit: true
  })
}

  // 多时间框架数据获取
  async fetchMultiTimeframeData() {
    const [tf1, tf4, tfD] = await Promise.all([
      this.fetchOHLCV('15m', 100),
      this.fetchOHLCV('1h', 100),
      this.fetchOHLCV('1d', 100)
    ])
    return { tf1, tf4, tfD }
  }

  async fetchOHLCV(tf, limit) {
    const ohlcv = await config.exchange.fetchOHLCV(config.symbol, tf, undefined, limit)
    return ohlcv.map(([t, o, h, l, c, v]) => ({ t, o, h, l, c, v }))
  }

  // 动态指标计算
  calculateDynamicIndicators(data) {
    // 多周期EMA
    const emas = config.emaPeriods.map(p => ({
      period: p,
      value: this.calculateEMA(data.tf1.map(d => d.c), p)
    }))

    // 波动率调整参数
    const atr = this.calculateATR(data.tf1)
    const volatilityRatio = atr / data.tf1.slice(-1)[0].c

    // 市场情绪指标
    const socialSentiment = this.fetchSocialSentiment()

    return { emas, atr, volatilityRatio, socialSentiment }
  }

  calculateEMA(prices, period) {
    return talib.execute({
      name: 'EMA',
      startIdx: 0,
      endIdx: prices.length - 1,
      inReal: prices,
      optInTimePeriod: period
    }).result.outReal.slice(-1)[0]
  }

  calculateATR(data) {
    const highs = data.map(d => d.h)
    const lows = data.map(d => d.l)
    const closes = data.map(d => d.c)
    return talib.execute({
      name: 'ATR',
      startIdx: 0,
      endIdx: data.length - 1,
      high: highs,
      low: lows,
      close: closes,
      optInTimePeriod: config.atrPeriod
    }).result.outReal.slice(-1)[0]
  }

  // 增强信号生成
  generateEnhancedSignal(indicators) {
    const price = indicators.currentPrice

      indicators.emas[0].value > indicators.emas[1].value &&

    const dynamicThreshold = 0.15 + (indicators.volatilityRatio * 100) * 0.1
    const slope = this.calculateEMASlope(indicators.emas[0].values)

    // 复合信号条件
    const buyCondition = 
      emaConditions &&
      slope > dynamicThreshold &&
      indicators.socialSentiment > 0.6 &&
      this.checkVolumeSpike()

    return { action: buyCondition ? 'LONG' : 'NEUTRAL' }
  }

  // 动态风险控制
  async executeTrade(signal) {
    if (this.checkRiskLimits()) return

    const balance = await config.exchange.fetchBalance()
    const positionSize = this.calculatePositionSize(balance.USDT.free)

    try {
      const order = await config.exchange.createOrder(
        config.symbol,
        'market',
        signal.action === 'LONG' ? 'buy' : 'sell',
        positionSize
      )
      
      this.setStopLoss(order)
      this.logTrade(order)
    } catch (error) {
      this.handleOrderError(error)
    }
  }

    const riskAmount = balance * config.riskPerTrade

  // 增强功能模块
  async fetchSocialSentiment() {
    const response = await axios.get('https://api.social-sentiment.com/doge')
    return response.data.sentimentIndex
  }

  checkVolumeSpike() {
    const volumes = data.tf1.map(d => d.v)
    const currentVol = volumes.slice(-1)[0]
    const avgVol = volumes.slice(-10).reduce((a,b)=>a+b)/10
    return currentVol > avgVol * 1.5
  }

  // 风险控制模块
  checkRiskLimits() {
    return this.dailyLoss >= config.maxDailyLoss || 
           this.consecutiveLosses >= 3
  }

}

// 交易系统运行器
class TradingSystem {
  constructor() {
    this.bot = new EnhancedDogeBot()
    this.interval = 900000 // 15分钟
  }

  async start() {
    setInterval(async () => {
      try {
        const data = await this.bot.fetchMultiTimeframeData()
        const indicators = this.bot.calculateDynamicIndicators(data)
        const signal = this.bot.generateEnhancedSignal(indicators)
        await this.bot.executeTrade(signal)
      } catch (error) {
        console.error('系统错误:', error)
      }
    }, this.interval)
  }
}

// 启动系统
const system = new TradingSystem()
system.start()

### 风险控制矩阵

| 风险类型        | 缓解措施                          | 触发阈值         |
|-----------------|-----------------------------------|------------------|
| 极端波动        | 动态停损+波动率限制               | ATR > 5%         |
| 流动性不足      | 最小交易量检查+滑点控制           | 订单簿深度 < 1%  |
| 系统故障        | 心跳监测+自动恢复机制             | 超时 > 30s       |
| 交易所异常      | 多平台对冲+API异常处理            | 错误码 >= 500    |
