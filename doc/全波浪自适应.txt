以下是一个基于Node.js、ccxt和tulind实现的狗狗币5分钟周期全波浪次数自适应策略的完整代码，包含波浪检测、动态风控和交易执行功能：

```javascript
const ccxt = require('ccxt');
const tulind = require('tulind');
const { setTimeout } = require('timers/promises');

// 策略配置
const config = {
  symbol: 'DOGE/USDT',
  timeframe: '5m',
  exchange: 'binance',
  atrPeriod: 14,          // ATR周期
  emaPeriod: 20,          // 趋势过滤EMA周期
  waveConfirmation: 3,    // 波浪确认次数
  volumeMultiplier: 2.0,  // 成交量阈值倍数
  maxLeverage: 10,        // 最大杠杆
  baseOrderSize: 0.1,     // 基础仓位比例
  riskFactor: 0.02        // 单笔风险系数
};

// 初始化交易所
const exchange = new ccxt.binance({
  apiKey: process.env.API_KEY,
  secret: process.env.API_SECRET,
  enableRateLimit: true,
  options: { defaultType: 'future' }
});

class WaveStrategy {
  constructor() {
    this.waveState = {
      currentWave: null,
      waveCount: 0,
      waveDirection: 'none'
    };
    this.position = null;
  }

  async run() {
    try {
      while (true) {
        const candles = await this.fetchCandles();
        if (!candles || candles.length < 100) {
          await setTimeout(5000);
          continue;
        }

        // 计算技术指标
        const [atr, ema] = await Promise.all([
          this.calculateATR(candles),
          this.calculateEMA(candles)
        ]);

        // 波浪分析
        const waveAnalysis = this.analyzeWaves(candles);
        this.updateWaveState(waveAnalysis);

        // 生成交易信号
        const signal = this.generateSignal(candles, atr, ema);

        // 执行交易逻辑
        if (signal.action !== 'hold') {
          await this.executeOrder(signal, candles[candles.length-1].close);
        }

        // 更新止损
        await this.updateTrailingStop(candles);

        await setTimeout(1000 * 60 * 5); // 5分钟间隔
      }
    } catch (error) {
      console.error('策略运行错误:', error);
    }
  }

  // 获取K线数据
  async fetchCandles() {
    try {
      const data = await exchange.fetchOHLCV(
        config.symbol,
        config.timeframe,
        undefined,
        100
      );
      return data.map(d => ({
        timestamp: d[0],
        open: d[1],
        high: d[2],
        low: d[3],
        close: d[4],
        volume: d[5]
      }));
    } catch (error) {
      console.error('获取K线失败:', error.message);
      return null;
    }
  }

  // 波浪分析核心逻辑
  analyzeWaves(candles) {
    const swingPoints = this.findSwingPoints(candles);
    const waveStatus = {
      isUpTrend: false,
      isDownTrend: false,
      waveCount: 0
    };

    if (swingPoints.highs.length >= 3 && swingPoints.lows.length >= 2) {
      const lastHighs = swingPoints.highs.slice(-3);
      const lastLows = swingPoints.lows.slice(-2);
      
      // 检测上升波浪
      waveStatus.isUpTrend = lastHighs[2].price > lastHighs[1].price && 
                            lastHighs[1].price > lastHighs[0].price &&
                            lastLows[1].price > lastLows[0].price;

      // 检测下降波浪
      waveStatus.isDownTrend = lastHighs[2].price < lastHighs[1].price && 
                              lastHighs[1].price < lastHighs[0].price &&
                              lastLows[1].price < lastLows[0].price;

      waveStatus.waveCount = this.calculateWaveCount(swingPoints);
    }

    return waveStatus;
  }

  // 寻找波动高低点
  findSwingPoints(candles) {
    const swingPoints = { highs: [], lows: [] };
    
    for (let i = 2; i < candles.length - 2; i++) {
      const window = candles.slice(i - 2, i + 3);
      const center = window[2];
      
      if (center.high === Math.max(...window.map(c => c.high)) {
        swingPoints.highs.push({
          index: i,
          price: center.high,
          timestamp: center.timestamp
        });
      }
      
      if (center.low === Math.min(...window.map(c => c.low)) {
        swingPoints.lows.push({
          index: i,
          price: center.low,
          timestamp: center.timestamp
        });
      }
    }
    
    return swingPoints;
  }

  // 生成交易信号
  generateSignal(candles, atr, ema) {
    const currentPrice = candles[candles.length-1].close;
    const lastCandle = candles[candles.length-1];
    const volumeValid = lastCandle.volume > 
      this.averageVolume(candles.slice(-5)) * config.volumeMultiplier;

    const signal = { action: 'hold', size: 0 };

    // 多头条件
    if (this.waveState.isUpTrend &&
        currentPrice > ema[ema.length-1] &&
        volumeValid &&
        this.waveState.waveCount >= config.waveConfirmation) {
      signal.action = 'long';
      signal.size = this.calculatePositionSize(atr[atr.length-1], currentPrice);
    }

    // 空头条件
    else if (this.waveState.isDownTrend &&
        currentPrice < ema[ema.length-1] &&
        volumeValid &&
        this.waveState.waveCount >= config.waveConfirmation) {
      signal.action = 'short';
      signal.size = this.calculatePositionSize(atr[atr.length-1], currentPrice);
    }

    return signal;
  }

  // 动态仓位计算
  calculatePositionSize(atrValue, price) {
    const riskAmount = await this.getAccountBalance() * config.riskFactor;
    const riskPerUnit = atrValue * price;
    return Math.min(riskAmount / riskPerUnit, config.baseOrderSize);
  }

  // 执行订单
  async executeOrder(signal, price) {
    try {
      if (this.position) await this.closePosition();

      const order = await exchange.createOrder(
        config.symbol,
        'market',
        signal.action,
        signal.size,
        undefined,
        { leverage: config.maxLeverage }
      );

      this.position = {
        side: signal.action,
        entryPrice: price,
        stopLoss: this.calculateStopLoss(signal.action, price, atr),
        takeProfit: this.calculateTakeProfit(signal.action, price),
        size: signal.size
      };

      console.log(`订单执行成功: ${signal.action} ${signal.size} @ ${price}`);
    } catch (error) {
      console.error('订单执行失败:', error);
    }
  }

  // 动态止损计算
  calculateStopLoss(side, price, atr) {
    const atrMultiple = this.waveState.waveCount > 3 ? 1.5 : 2.0;
    return side === 'long' 
      ? price - (atr * atrMultiple)
      : price + (atr * atrMultiple);
  }

  // 动态止盈计算
  calculateTakeProfit(side, price) {
    const riskReward = this.waveState.waveCount > 3 ? 2.5 : 3.0;
    return side === 'long'
      ? price + (price - this.position.stopLoss) * riskReward
      : price - (this.position.stopLoss - price) * riskReward;
  }

  // 更新移动止损
  async updateTrailingStop(currentPrice) {
    if (!this.position) return;

    // 多头移动止损逻辑
    if (this.position.side === 'long') {
      const newStop = currentPrice * 0.995; // 0.5%回撤触发
      this.position.stopLoss = Math.max(this.position.stopLoss, newStop);
    } 
    // 空头移动止损逻辑
    else {
      const newStop = currentPrice * 1.005;
      this.position.stopLoss = Math.min(this.position.stopLoss, newStop);
    }

    // 检查止损条件
    if ((this.position.side === 'long' && currentPrice <= this.position.stopLoss) ||
        (this.position.side === 'short' && currentPrice >= this.position.stopLoss)) {
      await this.closePosition();
    }
  }

  // 辅助函数
  async getAccountBalance() {
    const balance = await exchange.fetchBalance();
    return balance.total.USDT;
  }

  averageVolume(candles) {
    return candles.reduce((sum, c) => sum + c.volume, 0) / candles.length;
  }

  async calculateATR(candles) {
    const inputs = candles.map(c => [c.high, c.low, c.close]);
    return (await tulind.indicators.atr.indicator(inputs, [config.atrPeriod]))[0];
  }

  async calculateEMA(candles) {
    return (await tulind.indicators.ema.indicator(
      [candles.map(c => c.close)], 
      [config.emaPeriod]
    ))[0];
  }
}

// 启动策略
const strategy = new WaveStrategy();
strategy.run();
```

### 代码核心功能说明：

1. **多维度波浪检测**：
   - 通过`findSwingPoints`函数识别最近3个高点/低点
   - 使用三重波浪确认机制（需连续3个递增高点/递减低点）

2. **动态风险管理**：
   - 根据ATR值计算止损（波动大时放宽止损）
   - 仓位大小与账户余额和波动率挂钩（`calculatePositionSize`）
   - 移动止损机制（`updateTrailingStop`）

3. **趋势过滤系统**：
   - 使用EMA20作为趋势方向过滤器
   - 成交量必须达到5日均量的2倍才确认信号

4. **自适应参数调整**：
   - 波浪次数超过3次时降低止盈风险比
   - 根据波浪阶段动态调整ATR乘数
